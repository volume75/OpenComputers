-- Base64 (GitHub requires it)
---------------------------------------------------
local b64chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/'

local function base64(data)
    return ((data:gsub('.', function(x)
        local r,b='',x:byte()
        for i=8,1,-1 do r=r..(b%2^i-b%2^(i-1)>0 and '1' or '0') end
        return r
    end)..'0000'):gsub('%d%d%d?%d?%d?%d?', function(x)
        if (#x < 6) then return '' end
        local c=0
        for i=1,6 do
            c=c + (x:sub(i,i)=='1' and 2^(6-i) or 0)
        end
        return b64chars:sub(c+1,c+1)
    end)..({ '', '==', '=' })[#data%3+1])
end

---------------------------------------------------
-- Read file
---------------------------------------------------
local function readFile(path)
    local f = io.open(path, "rb")
    if not f then return nil end
    local data = f:read("*a")
    f:close()
    return data
end

---------------------------------------------------
-- Scan directory recursively
---------------------------------------------------
local function scan(dir, out)
    out = out or {}
    for file in fs.list(dir) do
        local full = fs.concat(dir, file)
        if fs.isDirectory(full) then
            scan(full, out)
        else
            table.insert(out, full)
        end
    end
    return out
end

---------------------------------------------------
-- Upload file to GitHub
---------------------------------------------------
local function uploadFile(localPath, remotePath)
    local data = readFile(localPath)
    if not data then
        print("Failed to read:", localPath)
        return
    end

    local url = "https://api.github.com/repos/" ..
        USER .. "/" .. REPO ..
        "/contents/" .. githubFolder .. "/" .. remotePath

    local payload = {
        message = "OC upload " .. remotePath,
        content = encoded,
        branch = BRANCH
    }

    local json = require("serialization").serialize(payload)

    local headers = {
        ["Authorization"] = "Bearer " .. TOKEN,
        ["Content-Type"] = "application/json",
        ["User-Agent"] = "OpenComputers"
    }

    local response = internet.request(url, json, headers, "PUT")

    local result = ""
    for chunk in response do
        result = result .. chunk
    end

    print("Uploaded:", remotePath)
end
